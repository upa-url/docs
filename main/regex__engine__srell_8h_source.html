45 bool init(std::string_view regex_str, bool ignore_case) {
46 // 4. If options's ignore case is true then set flags to "vi".
47 // 5. Otherwise set flags to "v"
48 const auto commonflags = srell::regex::ECMAScript | srell::regex::vmode |
49 srell::regex::quiet; // do not throw a srell::regex_error on errors
50 const auto flag = ignore_case
51 ? (commonflags | srell::regex::icase)
52 : commonflags;
53 return re_.assign(regex_str.data(), regex_str.length(), flag).ecode() == 0;
54 }

56 bool exec(std::string_view input, result& res) const {
57 return srell::regex_match(input.begin(), input.end(), res.arr_, re_);
58 }
59 bool test(std::string_view input) const {
60 return srell::regex_match(input.begin(), input.end(), re_);
61 }
