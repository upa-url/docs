28template <class StrT, enable_if_str_arg_t<StrT> = 0>
29constexpr std::size_t get_label_pos_by_index(StrT&& str_host, std::size_t index) {
30 const auto inp = make_str_arg(std::forward<StrT>(str_host));
31 const auto* first = inp.begin();
32 const auto* last = inp.end();
33 const auto* ptr = first;
34 for (; index > 0; --index) {
35 while (true) {
36 if (ptr == last)
37 return static_cast<std::size_t>(ptr - first);
38 const auto c = util::to_unsigned(*ptr);
39 if (c < 0x80) {
40 ++ptr; // advance to next char
41 if (c == '.')
42 break;
43 } else {
44 const auto cp = url_utf::read_utf_char(ptr, last).value;
45 // These code points are mapped to '.' in the IDNA mapping table
46 if (cp == 0x3002 || cp == 0xFF0E || cp == 0xFF61)
47 break;
48 }
49 }
50 }
51 return static_cast<std::size_t>(ptr - first);
52}

284 template <class StrT, enable_if_str_arg_t<StrT> = 0>
285 auto get_suffix_view(StrT&& str_host, option opt = PUBLIC_SUFFIX) const
286 -> std::basic_string_view<typename str_arg<str_arg_char_t<StrT>>::value_type> {
287 try {
288 const auto arg = make_str_arg(std::forward<StrT>(str_host));
289 const upa::url_host host{ arg };
290 if (host.type() == HostType::Domain) {
291 const auto res = get_host_suffix_info(host.name(), opt);
292 if (res) {
293 const auto pos = get_label_pos_by_index(arg, res.first_label_ind);
294 return { arg.data() + pos, arg.size() - pos };
295 }
296 }
297 }
298 catch (const upa::url_error&) {}
299 return {};
300 }

319private:
320 struct label_item {
321#ifdef __cpp_lib_generic_unordered_lookup
322 // https://en.cppreference.com/w/cpp/container/unordered_map/find
323 struct string_hash {
324 using hash_type = std::hash<std::string_view>;
325 using is_transparent = void;
326
327 std::size_t operator()(const char* str) const { return hash_type{}(str); }
328 std::size_t operator()(std::string_view str) const { return hash_type{}(str); }
329 std::size_t operator()(std::string const& str) const { return hash_type{}(str); }
330 };
331 using map_type = std::unordered_map<std::string, label_item, string_hash, std::equal_to<>>;
332#else
333 using map_type = std::unordered_map<std::string, label_item>;
334#endif
335
336 std::uint8_t code = 0;
337 std::unique_ptr<map_type> children;
338
339 bool operator==(const label_item& b) const {
340 return code == b.code && (
341 (!children && !b.children) ||
342 (children && b.children && *children == *b.children));
343 }
344 };
