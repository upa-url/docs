28template <class StrT, enable_if_str_arg_t<StrT> = 0>
29constexpr std::size_t get_label_pos_by_index(StrT&& str_host, std::size_t index) {
30 const auto inp = make_str_arg(std::forward<StrT>(str_host));
31 const auto* first = inp.begin();
32 const auto* last = inp.end();
33 const auto* ptr = first;
34 for (; index > 0; --index) {
35 while (true) {
36 if (ptr == last)
37 return static_cast<std::size_t>(ptr - first);
38 const auto c = util::to_unsigned(*ptr);
39 if (c < 0x80) {
40 ++ptr; // advance to next char
41 if (c == '.')
42 break;
43 } else {
44 const auto cp = url_utf::read_utf_char(ptr, last).value;
45 // These code points are mapped to '.' in the IDNA mapping table
46 if (cp == 0x3002 || cp == 0xFF0E || cp == 0xFF61)
47 break;
48 }
49 }
50 }
51 return static_cast<std::size_t>(ptr - first);
52}

283 template <class StrT, enable_if_str_arg_t<StrT> = 0>
284 auto get_suffix_view(StrT&& str_host, option opt = option::public_suffix) const
285 -> std::basic_string_view<typename str_arg<str_arg_char_t<StrT>>::value_type> {
286 try {
287 const auto arg = make_str_arg(std::forward<StrT>(str_host));
288 const upa::url_host host{ arg };
289 if (host.type() == HostType::Domain) {
290 const auto res = get_host_suffix_info(host.name(), opt);
291 if (res) {
292 const auto pos = get_label_pos_by_index(arg, res.first_label_ind);
293 return { arg.data() + pos, arg.size() - pos };
294 }
295 }
296 }
297 catch (const upa::url_error&) {}
298 return {};
299 }

319 struct label_item {
320#ifdef __cpp_lib_generic_unordered_lookup
321 // https://en.cppreference.com/w/cpp/container/unordered_map/find
322 struct string_hash {
323 using hash_type = std::hash<std::string_view>;
324 using is_transparent = void;
325
326 std::size_t operator()(const char* str) const { return hash_type{}(str); }
327 std::size_t operator()(std::string_view str) const { return hash_type{}(str); }
328 std::size_t operator()(std::string const& str) const { return hash_type{}(str); }
329 };
330 using map_type = std::unordered_map<std::string, label_item, string_hash, std::equal_to<>>;
331#else
332 using map_type = std::unordered_map<std::string, label_item>;
333#endif
334
335 std::uint8_t code = 0;
336 std::unique_ptr<map_type> children;
337
338 bool operator==(const label_item& b) const {
339 return code == b.code && (
340 (!children && !b.children) ||
341 (children && b.children && *children == *b.children));
342 }
343 };
