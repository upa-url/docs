89 template <class StrT, enable_if_str_arg_t<StrT> = 0>
90 explicit url_host(StrT&& str) {
91 host_out out(*this);
92
93 const auto inp = make_str_arg(std::forward<StrT>(str));
94 const auto res = host_parser::parse_host(inp.begin(), inp.end(), false, out);
95 if (res != validation_errc::ok)
96 throw url_error(res, "Host parse error");
97 }
98
100 ~url_host() = default;
101
105 [[nodiscard]] HostType type() const {
106 return type_;
107 }
108
112 [[nodiscard]] string_view name() const UPA_LIFETIMEBOUND {
113 return host_str_;
114 }
115
119 [[nodiscard]] std::string to_string() const {
120 return host_str_;
121 }
122
123private:
124 class host_out : public host_output {
125 public:
126 explicit host_out(url_host& host)
127 : host_(host)
128 {}
129 std::string& hostStart() override {
130 return host_.host_str_;
131 }
132 void hostDone(HostType ht) override {
133 host_.type_ = ht;
134 }
135 private:
136 url_host& host_; // NOLINT(cppcoreguidelines-avoid-const-or-ref-data-members)
137 };
138
139 // members
140 std::string host_str_;
141 HostType type_ = HostType::Empty;
142};

176template <class CharT, class StrT, enable_if_str_arg_t<StrT> = 0>
177inline bool domain_to_unicode(std::basic_string<CharT>& output, StrT&& input,
178 bool be_strict = false, bool is_input_ascii = false)
179{
180 const auto inp = make_str_arg(std::forward<StrT>(input));
181 if constexpr (std::is_same_v<CharT, char32_t>) {
182 return idna::domain_to_unicode(output, inp.begin(), inp.end(), be_strict, is_input_ascii);
183 } else {
184 std::u32string domain;
185 const bool res = idna::domain_to_unicode(domain, inp.begin(), inp.end(), be_strict, is_input_ascii);
186 if constexpr (sizeof(CharT) == sizeof(char)) {
187 // CharT is char8_t, or char
188 for (auto cp : domain)
189 url_utf::append_utf8<std::basic_string<CharT>, detail::append_to_string>(cp, output);
190 } else if constexpr (sizeof(CharT) == sizeof(char16_t)) {
191 // CharT is char16_t, or wchar_t (Windows)
192 for (auto cp : domain)
193 url_utf::append_utf16(cp, output);
194 } else if constexpr (sizeof(CharT) == sizeof(char32_t)) {
195 // CharT is wchar_t (non Windows)
196 util::append(output, domain);
197 } else {
198 static_assert(util::false_v<CharT>, "unsupported output character type");
199 }
200 return res;
201 }
202}

205// https://url.spec.whatwg.org/#concept-host-parser

207template <typename CharT>
208inline validation_errc host_parser::parse_host(const CharT* first, const CharT* last, bool is_opaque, host_output& dest) {
209 using UCharT = std::make_unsigned_t<CharT>;
210
211 // 1. Non-"file" special URL's cannot have an empty host.
212 // 2. For "file" URL's empty host is set in the file_host_state 1.2
213 // https://url.spec.whatwg.org/#file-host-state
214 // 3. Non-special URLs can have empty host and it will be set here.
215 if (first == last) {
216 // https://github.com/whatwg/url/issues/79
217 // https://github.com/whatwg/url/pull/189
218 // set empty host
219 dest.hostStart();
220 dest.hostDone(HostType::Empty);
221 return is_opaque ? validation_errc::ok : validation_errc::host_missing;
222 }
223 assert(first < last);
224
225 if (*first == '[') {
226 if (*(last - 1) == ']') {
227 return parse_ipv6(first + 1, last - 1, dest);
228 }
229 // 1.1. If input does not end with U+005D (]), IPv6-unclosed
230 // validation error, return failure.
231 return validation_errc::ipv6_unclosed;
232 }
233
234 if (is_opaque)
235 return parse_opaque_host(first, last, dest);
236
237 // Is ASCII domain?
238 const auto ptr = std::find_if_not(first, last, detail::is_ascii_domain_char<CharT>);
239 if (ptr == last) {
240 if (!util::has_xn_label(first, last)) {
241 // Fast path for ASCII domain
242
243 // If asciiDomain ends in a number, return the result of IPv4 parsing asciiDomain
244 if (hostname_ends_in_a_number(first, last))
245 return parse_ipv4(first, last, dest);
246
247 if (dest.need_save()) {
248 // Return asciiDomain lower cased
249 std::string& str_host = dest.hostStart();
250 util::append_ascii_lowercase(str_host, first, last);
251 dest.hostDone(HostType::Domain);
252 }
253 return validation_errc::ok;
254 }
255 } else if (static_cast<UCharT>(*ptr) < 0x80 && *ptr != '%') {
256 // NFC normalizes U+003C (<), U+003D (=), U+003E (>) characters if they precede
257 // U+0338. Therefore, no errors are reported here for forbidden < and > characters
258 // if there is a possibility to normalize them.
259 if (!(*ptr >= 0x3C && *ptr <= 0x3E && ptr + 1 < last && static_cast<UCharT>(ptr[1]) >= 0x80))
260 // 7. If asciiDomain contains a forbidden domain code point, domain-invalid-code-point
261 // validation error, return failure.
262 return validation_errc::domain_invalid_code_point;
263 }
264
265 std::string buff_ascii;
266
267 const auto pes = std::find(ptr, last, '%');
268 if (pes == last) {
269 // Input is ASCII if ptr == last
270 if (!idna::domain_to_ascii(buff_ascii, first, last, false, ptr == last))
271 return validation_errc::domain_to_ascii;
272 } else {
273 // Input for domain_to_ascii
274 simple_buffer<char32_t> buff_uc;
275
276 // copy ASCII chars
277 for (auto it = first; it != ptr; ++it) {
278 const auto uch = static_cast<UCharT>(*it);
279 buff_uc.push_back(static_cast<char32_t>(uch));
280 }
281
282 // Let buff_uc be the result of running UTF-8 decode (to UTF-16) without BOM
283 // on the percent decoding of UTF-8 encode on input
284 for (auto it = ptr; it != last;) {
285 const auto uch = static_cast<UCharT>(*it++);
286 if (uch < 0x80) {
287 if (uch != '%') {
288 buff_uc.push_back(static_cast<char32_t>(uch));
289 continue;
290 }
291 // uch == '%'
292 unsigned char uc8; // NOLINT(cppcoreguidelines-init-variables)
293 if (detail::decode_hex_to_byte(it, last, uc8)) {
294 if (uc8 < 0x80) {
295 buff_uc.push_back(static_cast<char32_t>(uc8));
296 continue;
297 }
298 // percent encoded utf-8 sequence
299 // TODO: gal po vieną code_point, tuomet užtektų utf-8 buferio vienam simboliui
300 simple_buffer<char> buff_utf8;
301 buff_utf8.push_back(static_cast<char>(uc8));
302 while (it != last && *it == '%') {
303 ++it; // skip '%'
304 if (!detail::decode_hex_to_byte(it, last, uc8))
305 uc8 = '%';
306 buff_utf8.push_back(static_cast<char>(uc8));
307 }
308 // utf-8 to utf-32
309 const auto* last_utf8 = buff_utf8.data() + buff_utf8.size();
310 for (const auto* it_utf8 = buff_utf8.data(); it_utf8 < last_utf8;)
311 buff_uc.push_back(url_utf::read_utf_char(it_utf8, last_utf8).value);
312 //buff_utf8.clear();
313 continue;
314 }
315 // detected an invalid percent-encoding sequence
316 buff_uc.push_back('%');
317 } else { // uch >= 0x80
318 --it;
319 buff_uc.push_back(url_utf::read_utf_char(it, last).value);
320 }
321 }
322 if (!idna::domain_to_ascii(buff_ascii, buff_uc.begin(), buff_uc.end()))
323 return validation_errc::domain_to_ascii;
324 }
325
326 if (detail::contains_forbidden_domain_char(buff_ascii.data(), buff_ascii.data() + buff_ascii.size())) {
327 // 7. If asciiDomain contains a forbidden domain code point, domain-invalid-code-point
328 // validation error, return failure.
329 return validation_errc::domain_invalid_code_point;
330 }
331
332 // If asciiDomain ends in a number, return the result of IPv4 parsing asciiDomain
333 if (hostname_ends_in_a_number(buff_ascii.data(), buff_ascii.data() + buff_ascii.size()))
334 return parse_ipv4(buff_ascii.data(), buff_ascii.data() + buff_ascii.size(), dest);
335
336 if (dest.need_save()) {
337 // Return asciiDomain
338 std::string& str_host = dest.hostStart();
339 str_host.append(buff_ascii);
340 dest.hostDone(HostType::Domain);
341 }
342 return validation_errc::ok;
343}

345// The opaque-host parser
346// https://url.spec.whatwg.org/#concept-opaque-host-parser

348template <typename CharT>
349inline validation_errc host_parser::parse_opaque_host(const CharT* first, const CharT* last, host_output& dest) {
350 // 1. If input contains a forbidden host code point, host-invalid-code-point
351 // validation error, return failure.
352 if (detail::contains_forbidden_host_char(first, last))
353 return validation_errc::host_invalid_code_point;
354
355 // TODO-WARN:
356 // 2. If input contains a code point that is not a URL code point and not U+0025 (%),
357 // invalid-URL-unit validation error.
358 // 3. If input contains a U+0025 (%) and the two code points following it are not ASCII hex digits,
359 // invalid-URL-unit validation error.
360
361 if (dest.need_save()) {
362 std::string& str_host = dest.hostStart();
363
364 //TODO: UTF-8 percent encode it using the C0 control percent-encode set
365 //detail::append_utf8_percent_encoded(first, last, detail::CHAR_C0_CTRL, str_host);
366 using UCharT = std::make_unsigned_t<CharT>;
367
368 const CharT* pointer = first;
369 while (pointer < last) {
370 // UTF-8 percent encode c using the C0 control percent-encode set (U+0000 ... U+001F and >U+007E)
371 const auto uch = static_cast<UCharT>(*pointer);
372 if (uch >= 0x7f) {
373 // invalid utf-8/16/32 sequences will be replaced with 0xfffd
374 detail::append_utf8_percent_encoded_char(pointer, last, str_host);
375 } else {
376 // Just append the 7-bit character, percent encoding C0 control chars
377 const auto uc = static_cast<unsigned char>(uch);
378 if (uc <= 0x1f)
379 detail::append_percent_encoded_byte(uc, str_host);
380 else
381 str_host.push_back(uc);
382 ++pointer;
383 }
384 }
385
386 dest.hostDone(str_host.empty() ? HostType::Empty : HostType::Opaque);
387 }
388 return validation_errc::ok;
389}

391template <typename CharT>
392inline validation_errc host_parser::parse_ipv4(const CharT* first, const CharT* last, host_output& dest) {
393 uint32_t ipv4; // NOLINT(cppcoreguidelines-init-variables)
394
395 const auto res = ipv4_parse(first, last, ipv4);
396 if (res == validation_errc::ok && dest.need_save()) {
397 std::string& str_ipv4 = dest.hostStart();
398 ipv4_serialize(ipv4, str_ipv4);
399 dest.hostDone(HostType::IPv4);
400 }
401 return res;
402}

404template <typename CharT>
405inline validation_errc host_parser::parse_ipv6(const CharT* first, const CharT* last, host_output& dest) {
406 uint16_t ipv6addr[8];
407
408 const auto res = ipv6_parse(first, last, ipv6addr);
409 if (res == validation_errc::ok && dest.need_save()) {
410 std::string& str_ipv6 = dest.hostStart();
411 str_ipv6.push_back('[');
412 ipv6_serialize(ipv6addr, str_ipv6);
413 str_ipv6.push_back(']');
414 dest.hostDone(HostType::IPv6);
415 }
416 return res;
417}
