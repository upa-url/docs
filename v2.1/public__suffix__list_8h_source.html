31template <class StrT, enable_if_str_arg_t<StrT> = 0>
32constexpr std::size_t get_label_pos_by_index(StrT&& str_host, std::size_t index) {
33 const auto inp = make_str_arg(std::forward<StrT>(str_host));
34 const auto* first = inp.begin();
35 const auto* last = inp.end();
36 const auto* ptr = first;
37 for (; index > 0; --index) {
38 while (true) {
39 if (ptr == last)
40 return static_cast<std::size_t>(ptr - first);
41 const auto c = util::to_unsigned(*ptr);
42 if (c < 0x80) {
43 ++ptr; // advance to next char
44 if (c == '.')
45 break;
46 } else {
47 const auto cp = url_utf::read_utf_char(ptr, last).value;
48 // These code points are mapped to '.' in the IDNA mapping table
49 if (cp == 0x3002 || cp == 0xFF0E || cp == 0xFF61)
50 break;
51 }
52 }
53 }
54 return static_cast<std::size_t>(ptr - first);
55}

286 template <class StrT, enable_if_str_arg_t<StrT> = 0>
287 auto get_suffix_view(StrT&& str_host, option opt = option::public_suffix) const
288 -> std::basic_string_view<typename str_arg<str_arg_char_t<StrT>>::value_type> {
289 try {
290 const auto arg = make_str_arg(std::forward<StrT>(str_host));
291 const upa::url_host host{ arg };
292 if (host.type() == HostType::Domain) {
293 const auto res = get_host_suffix_info(host.name(), opt);
294 if (res) {
295 const auto pos = get_label_pos_by_index(arg, res.first_label_ind);
296 return { arg.data() + pos, arg.size() - pos };
297 }
298 }
299 }
300 catch (const upa::url_error&) {}
301 return {};
302 }

329#ifdef __cpp_lib_generic_unordered_lookup
330 // https://en.cppreference.com/w/cpp/container/unordered_map/find
331 struct string_hash {
332 using hash_type = std::hash<std::string_view>;
333 using is_transparent = void;
334
335 std::size_t operator()(const char* str) const { return hash_type{}(str); }
336 std::size_t operator()(std::string_view str) const { return hash_type{}(str); }
337 std::size_t operator()(std::string const& str) const { return hash_type{}(str); }
338 };
339 using map_type = std::unordered_map<std::string, label_item, string_hash, std::equal_to<>>;
340#else
341 using map_type = std::unordered_map<std::string, label_item>;
342#endif
343
344 std::uint8_t code = 0;
345 std::unique_ptr<map_type> children;
346
347 bool operator==(const label_item& b) const {
348 return code == b.code && (
349 (!children && !b.children) ||
350 (children && b.children && *children == *b.children));
351 }
